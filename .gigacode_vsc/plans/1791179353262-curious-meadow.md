```python
def make_lags(prices, lags=(1, 2, 3, 5, 10, 22)):
    lag_list = []
    price_list = []
    for i in range(max_lag, len(prices)):
        lag_row = [prices[i - lag] for lag in lags]
        lag_list.append(lag_row)
        price_list.append(prices[i])
    return lag_list, price_list
```

**Станет:**
```python
def make_lags(prices, dates, lags=(1, 2, 3, 5, 10, 22)):
    """Возвращает словарь {date: {'lags': [...], 'price': value}}."""
    max_lag = max(lags)
    data = {}
    for i in range(max_lag, len(prices)):
        data[dates[i]] = {
            'lags': [prices[i - lag] for lag in lags],
            'price': prices[i]
        }
    return data
```
